    [Command("pause")]
    [Description("Pauses the player.")]
    internal async ValueTask<Result> PauseAsync()
    {
        QueuedLavalinkPlayer? player = await GetPlayerAsync();

        if (player is null)
        {
            return Result.Success; // Error message is already sent in GetPlayerAsync.
        }

        if (player.State is PlayerState.Paused)
        {
            return (Result)await feedback.SendContextualErrorAsync("The player is already paused.", ct: CancellationToken);
        }

        await player.PauseAsync(CancellationToken);
        return (Result)await feedback.SendContextualInfoAsync($"Paused {FormatTrackMessage(player.CurrentTrack!)}", ct: CancellationToken);
    }

    [Command("skip")]
    [Description("Skips the current track.")]
    internal async ValueTask<Result> SkipAsync([Description("The number of tracks to skip.")] int amount = 1)
    {
        QueuedLavalinkPlayer? player = await GetPlayerAsync();

        if (player is null)
        {
            return Result.Success; // Error message is already sent in GetPlayerAsync.
        }

        if (player.CurrentTrack is null)
        {
            return (Result)await feedback.SendContextualErrorAsync("There isn't any currently playing track to skip.", ct: CancellationToken);
        }

        if (player.Queue.IsEmpty)
        {
            return (Result)await feedback.SendContextualErrorAsync("There isn't any queued track to skip.", ct: CancellationToken);
        }

        await player.SkipAsync(amount, CancellationToken);
        return (Result)await feedback.SendContextualInfoAsync($"Skipped {FormatTrackMessage(player.CurrentTrack!)}", ct: CancellationToken);
    }

    [Command("list")]
    [Description("Lists every queued track.")]
    internal async ValueTask<Result> ListAsync()
    {
        QueuedLavalinkPlayer? player = await GetPlayerAsync();

        if (player is null)
        {
            return Result.Success; // Error message is already sent in GetPlayerAsync.
        }

        if (player.Queue.Count == 1)
        {
            return (Result)await feedback.SendContextualInfoAsync($"Now Playing: {FormatTrackMessage(player.CurrentTrack!)}", ct: CancellationToken);
        }
        else
        {
            for (int i = 2; i < player.Queue.Count; i++)
            {
                LavalinkTrack track = player.Queue[i].Track!;
                _trackListBuilder.AppendLine($"{i}: [{track.Title}]({track.Uri})");
            }

            try
            {
                return (Result)await feedback.SendContextualEmbedAsync
                (
                    new EmbedBuilder()
                        .WithDescription
                        (
                            $"Now Playing: {FormatTrackMessage(player.CurrentTrack!)}" +
                            Environment.NewLine +
                            _trackListBuilder.ToString()
                        )
                        .Build().Entity,
                    ct: CancellationToken
                );
            }
            finally
            {
                _trackListBuilder.Clear();
            }
        }
    }

    [Command("volume")]

    }

    [Command("shuffle")]
    [Description("Shuffles the queue.")]
    internal async ValueTask<Result> ShuffleAsync()
    {
        QueuedLavalinkPlayer? player = await GetPlayerAsync(connectToVoiceChannel: false);

        if (player is null)
        {
            return Result.Success; // Error message is already sent in GetPlayerAsync.
        }

        if (player.Queue.Count <= 1)
        {
            return (Result)await feedback.SendContextualErrorAsync("There is only a single track in the queue.", ct: CancellationToken);
        }

        await player.Queue.ShuffleAsync();
        return (Result)await feedback.SendContextualInfoAsync("Queue shuffled.", ct: CancellationToken);
    }

    [Command("clear")]
    [Description("Clears the queue.")]
    internal async ValueTask<Result> ClearQueueAsync()
    {
        QueuedLavalinkPlayer? player = await GetPlayerAsync(connectToVoiceChannel: false);

        if (player is null)
        {
            return Result.Success; // Error message is already sent in GetPlayerAsync.
        }

        if (player.Queue.Count < 0)
        {
            return (Result)await feedback.SendContextualErrorAsync("No queue exists in order to clear.", ct: CancellationToken);
        }

        await player.Queue.ClearAsync();
        return (Result)await feedback.SendContextualInfoAsync("Queue cleared.", ct: CancellationToken);
    }

    private async ValueTask<QueuedLavalinkPlayer?> GetPlayerAsync(bool connectToVoiceChannel = true)
    {
        if (CancellationToken.IsCancellationRequested || !interactionCtx.TryGetGuildID(out Snowflake guildId))
        {
            return null;
        }

        PlayerResult<QueuedLavalinkPlayer> result = await audio.Players.RetrieveAsync
        (
            interactionCtx,
            PlayerFactory.Queued,
            new PlayerRetrieveOptions(connectToVoiceChannel ? PlayerChannelBehavior.Join : PlayerChannelBehavior.None),
            CancellationToken
        );

        if (!result.IsSuccess)
        {
            await feedback.SendContextualErrorAsync(result.Status switch
            {
                PlayerRetrieveStatus.UserNotInVoiceChannel => "You are not connected to a voice channel.",
                PlayerRetrieveStatus.BotNotConnected => "I am currently not connected to a voice channel.",
                _ => "An unknown error has occured."

            }, ct: CancellationToken);

            return null;
        }

        return result.Player;
    }

    private string FormatTrackMessage(LavalinkTrack track)
        => $"[{track?.Title}]({track?.Uri}).";
